\section*{General Information}
\begin{itemize}
    \item \textbf{Set Description:} NNPDF3.0 NLO global fit, $\alpha_s(M_Z) = 0.118$. \\
    mem=0 $\Rightarrow$ average on replicas; mem=1--100 $\Rightarrow$ PDF replicas.
    \item \textbf{Set Index:} 260000
    \item \textbf{Authors:} NNPDF Collaboration. R.D. Ball, V. Bertone, S. Carrazza, C.S. Deans, L. Del Debbio, S. Forte, A. Guffanti, N.P. Hartland, J.I. Latorre, J. Rojo, M. Ubiali
    \item \textbf{Reference:} \href{https://arxiv.org/abs/1410.8849}{arXiv:1410.8849}
    \item \textbf{Format:} lhagrid1
    \item \textbf{Data Version:} 2
\end{itemize}

\section*{PDF Characteristics}
\begin{itemize}
    \item \textbf{Number of Members:} 101 (1 central + 100 replicas)
    \item \textbf{Particle:} proton (PDG ID = 2212)
    \item \textbf{Flavors included:} $[-5, -4, -3, -2, -1, 1, 2, 3, 4, 5, 21]$ \\
    (b, c, s, d, u quarks and antiquarks, plus gluon)
    \item \textbf{QCD Order:} Next-to-Leading Order (NLO)
    \item \textbf{Flavor Scheme:} variable flavor number scheme
    \item \textbf{Number of Active Flavors:} 5
    \item \textbf{Error Type:} replicas (Monte Carlo)
\end{itemize}

\section*{Kinematic Range}
\begin{itemize}
    \item $x$ range: $1.0 \times 10^{-9}$ to $1.0$
    \item $Q$ range: $1.0$ GeV to $10^{5}$ GeV
\end{itemize}

\section*{Usage Notes}
\begin{itemize}
    \item Member 0 corresponds to the average (central) PDF.
    \item Members 1--100 are Monte Carlo replicas for uncertainty estimation.
    \item Suitable for collider physics applications over a wide kinematic range.
    \item $\alpha_s$ values are internally interpolated by LHAPDF.
\end{itemize}
